void ACombatVehicle::CalculateDriftForces(float DeltaTime)
{
  // Get local velocity relative to actor rotation
  FVector LocalVel = GetActorTransform().InverseTransformVector(Velocity);

  // Calculate slip angle based on lateral velocity
  float SlipAngle = FMath::Atan2(LocalVel.Y, LocalVel.X);

  // Apply counter-force based on friction curve
  float GripFactor = TireFrictionCurve->GetFloatValue(SlipAngle);
  AddForce(GetActorRightVector() * -LocalVel.Y * GripFactor);
}
